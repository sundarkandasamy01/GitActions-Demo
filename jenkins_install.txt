pipeline {
  agent any                      // where to run
  environment { APP_VERSION = '1.0.0' }   // shared vars
  stages {
    stage('Build') {
      steps { echo "Compiling v${APP_VERSION}..." }
    }
    stage('Unit Test') {
      steps { echo 'Running unit tests...' }
    }
    stage('Package') {
      steps { echo "Packaging app-v${APP_VERSION}.zip" }
    }
  }
  post {                         // after all stages
    success { echo 'Pipeline succeeded cleanly!' }
  }
}
